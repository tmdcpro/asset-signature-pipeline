def hurst_exponent(price_series, max_lag=100): ts = np.log(price_series.dropna().values) lags = range(2, min(max_lag, len(ts)//2)) tau = [np.std(ts[l:] - ts[:-l]) for l in lags] poly = np.polyfit(np.log(list(lags)), np.log(tau), 1) return poly[0] * 2 def garch_params(returns): model = arch_model(returns * 100, vol="Garch", p=1, q=1) res = model.fit(disp="off") return (res.params["omega"], res.params["alpha[1]"], res.params["beta[1]"])
